  /* Works out an order from { skuId: quantity }. Used by the basket in the
     browser AND by the checkout server, so the two can never disagree.
     Free shipping (a set in the order, or items over freeOver) covers
     postage, but an oversized torch bought on its own always pays its box. */
  function quote(items, delivery, gift) {
    var S = skus(), lines = [], n = 0, sum = 0, oz = 0;
    var torches = 0, box = 0, oversized = 0, hasSet = false;
    Object.keys(items || {}).forEach(function (k) {
      var s = S[k], q = Math.floor(Number(items[k]));
      if (!s || !(q > 0) || q > 99) return;
      lines.push({ sku:k, item:s, qty:q });
      n += q; sum += s.price * q; oz += s.oz * q;
      if (s.set) { hasSet = true; return; }
      if (s.box > 0) { torches += q; if (s.box > box) box = s.box; }
      if (s.box === 2) oversized += q;
    });
    var pickup = delivery === "pickup" && SHIPPING.pickup;
    var ship = 0, free = false, oversizeOnly = false;
    if (n > 0 && !pickup) {
      if (hasSet || (SHIPPING.freeOver && sum >= SHIPPING.freeOver)) {
        if (oversized > 0) { ship = (SHIPPING.boxCharge[2] || 0) + SHIPPING.extraTorch * (oversized - 1); oversizeOnly = true; }
        else free = true;
      } else {
        ship = tierFor(oz).price;
        if (torches > 0) ship += (SHIPPING.boxCharge[box] || 0) + SHIPPING.extraTorch * (torches - 1);
      }
    }
    var giftFee = gift && n > 0 ? GIFT.price : 0;
    return { lines:lines, n:n, sum:sum, oz:oz, torches:torches, ship:ship, free:free, oversizeOnly:oversizeOnly,
             pickup:pickup, hasSet:hasSet, gift:giftFee, grand:sum + ship + giftFee,
             toFree: (!pickup && !hasSet && SHIPPING.freeOver && sum < SHIPPING.freeOver) ? SHIPPING.freeOver - sum : 0 };
  }

  function usage(items) {
    var S = skus(), u = {}, c = clean(items);
    Object.keys(c).forEach(function (k) {

      Object.keys(uses).forEach(function (key) { u[key] = (u[key] || 0) + uses[key] * c[k]; });
    });
    return u;

  /* sold = { stockKey: units sold this batch } -> what's left of each */

    var r = {};

      r[s.key] = s.count == null ? null : Math.max(0, s.count - ((sold && sold[s.key]) || 0));
    });
    return r;
  }
  /* First thing in a basket that there isn't enough of, or null. */
  function shortfall(items, rem) {
    var u = usage(items);
    for (var i = 0; i < STOCK.length; i++) {
      var key = STOCK[i].key;
      if (u[key] && rem[key] != null && u[key] > rem[key]) return { key:key, label:STOCK[i].label, left:rem[key] };
    }
    return null;
  }
  function canAdd(sku, items, rem) {
    var next = clean(items); next[sku] = (next[sku] || 0) + 1;
    return !shortfall(next, rem);
  }
  /* How many of one sku could be made from what's left (Infinity if untracked). */
  function unitsLeft(sku, rem) {
    var uses = (skus()[sku] || {}).uses || {}, best = Infinity;
    Object.keys(uses).forEach(function (key) {
      if (rem[key] != null) best = Math.min(best, Math.floor(rem[key] / uses[key]));
    });
    return best;
  }
  /* Compact form for Stripe metadata: "p0s4b1:1,p2:2" and back. */
  function packItems(items) {
    var c = clean(items);
    return Object.keys(c).map(function (k) { return k + ":" + c[k]; }).join(",");
  }
  function unpackItems(str) {
    var out = {};
    String(str || "").split(",").forEach(function (pair) {
      var p = pair.split(":"); if (p.length === 2) out[p[0]] = Number(p[1]);
    });
    return clean(out);
  }
