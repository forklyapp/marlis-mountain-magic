  var _skus = null;
  function skus() {
    if (_skus) return _skus;
    var S = {};
    BUNDLES.forEach(function (b, i) {

      if (p.sizes) p.sizes.forEach(function (s, j) {

        blends.forEach(function (b, k) {

      else S["p" + i] = { name:p.name, size:null, price:p.price, oz:p.oz || 0, box:0 };

    _skus = S;
    return S;

  /* Works out an order from { skuId: quantity }. Used by the basket in the
     browser AND by the checkout server, so the two can never disagree.   */
  function tierFor(oz) {
    for (var t = 0; t < SHIPPING.tiers.length; t++) {
      if (SHIPPING.tiers[t].maxOz === null || oz <= SHIPPING.tiers[t].maxOz) return SHIPPING.tiers[t];
    }
    return SHIPPING.tiers[SHIPPING.tiers.length - 1];
  }

  /* Works out an order from { skuId: quantity }. Used by the basket in the
     browser AND by the checkout server, so the two can never disagree.
     Free shipping (a set in the order, or over freeOver) covers postage,
     but an oversized torch bought on its own always pays its box charge. */
  function quote(items, delivery) {
    var S = skus(), lines = [], n = 0, sum = 0, oz = 0;
    var torches = 0, box = 0, oversized = 0, hasSet = false;
    Object.keys(items || {}).forEach(function (k) {
      var s = S[k], q = Math.floor(Number(items[k]));
      if (!s || !(q > 0) || q > 99) return;
      lines.push({ sku:k, item:s, qty:q });
      n += q; sum += s.price * q; oz += s.oz * q;
      if (s.set) { hasSet = true; return; }
      if (s.box > 0) { torches += q; if (s.box > box) box = s.box; }
      if (s.box === 2) oversized += q;
    });
    var pickup = delivery === "pickup" && SHIPPING.pickup;
    var ship = 0, free = false, oversizeOnly = false;
    if (n > 0 && !pickup) {
      if (hasSet || (SHIPPING.freeOver && sum >= SHIPPING.freeOver)) {
        if (oversized > 0) { ship = (SHIPPING.boxCharge[2] || 0) + SHIPPING.extraTorch * (oversized - 1); oversizeOnly = true; }
        else free = true;
      } else {
        ship = tierFor(oz).price;
        if (torches > 0) ship += (SHIPPING.boxCharge[box] || 0) + SHIPPING.extraTorch * (torches - 1);
      }
    }
    return { lines:lines, n:n, sum:sum, oz:oz, torches:torches, ship:ship, free:free, oversizeOnly:oversizeOnly,
             pickup:pickup, grand:sum + ship };
  }

    skus:skus, quote:quote
